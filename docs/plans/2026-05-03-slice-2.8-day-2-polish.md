```tsx
import Link from "next/link";
import type { ReactNode } from "react";

interface EmptyStateProps {
  headline: string;
  description?: string;
  cta?:
    | { href: string; label: string }
    | { onClick: () => void; label: string };
  icon?: ReactNode;
}

export function EmptyState({
  headline,
  description,
  cta,
  icon,
}: EmptyStateProps) {
  return (
    <div className="rounded-[var(--radius-xl)] border border-[var(--color-hairline)] bg-[var(--color-canvas)] py-16 text-center">
      {icon && (
        <div className="mx-auto mb-4 flex h-12 w-12 items-center justify-center rounded-full bg-[var(--color-surface-soft)] text-[var(--color-muted)]">
          {icon}
        </div>
      )}
      <h3 className="text-lg font-semibold text-[var(--color-ink)]">
        {headline}
      </h3>
      {description && (
        <p className="mx-auto mt-2 max-w-md text-sm text-[var(--color-body)]">
          {description}
        </p>
      )}
      {cta && "href" in cta && (
        <Link
          href={cta.href}
          className="mt-6 inline-flex items-center justify-center rounded-[var(--radius-pill)] bg-[var(--color-primary)] px-6 py-2 text-sm font-semibold text-[var(--color-on-primary)] hover:bg-[var(--color-primary-active)]"
        >
          {cta.label}
        </Link>
      )}
      {cta && "onClick" in cta && (
        <button
          onClick={cta.onClick}
          className="mt-6 inline-flex items-center justify-center rounded-[var(--radius-pill)] bg-[var(--color-primary)] px-6 py-2 text-sm font-semibold text-[var(--color-on-primary)] hover:bg-[var(--color-primary-active)]"
        >
          {cta.label}
        </button>
      )}
    </div>
  );
}
```

## Why This Slice Matters

This slice is the difference between a thesis demo that _plays_ and one that _embarrasses_. Day 2 ships the system's three thesis-defining features: profile scoring (2.5), match scoring on apply (2.6), and bias detection (2.7). 2.8 makes sure the surface around them — the loading states, the empty states, the mobile layouts, the tiny "AI Suggested" badge that _should_ appear next to a prefilled phone number — doesn't undermine those features.

A demo where the candidate hits Apply and sees a blank screen for 15 seconds (instead of the AI Shimmer playing) reads as broken even if the AI is working perfectly. A demo on a recruiter's iPad where the bias modal won't fit reads as half-built. 2.8 closes those gaps.
