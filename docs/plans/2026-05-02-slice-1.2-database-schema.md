**Goal:** Establish the full AuraHire database — all 15 tables in `packages/db/src/schema.ts` (Drizzle), generate SQL migration, apply schema + RLS policies to the Supabase project (`fzjvalmouygmmnrgpgtg`) directly via Supabase MCP, seed the default `scoring_config` row, and wire a Drizzle DB client provider into `apps/api`. **Step 1:** Write file:

```ts
import { Global, Module } from "@nestjs/common";
import { ConfigService } from "@nestjs/config";
import { drizzle, PostgresJsDatabase } from "drizzle-orm/postgres-js";
import postgres from "postgres";
import * as schema from "@aurahire/db";

export const DRIZZLE_CLIENT = Symbol("DRIZZLE_CLIENT");
export type DrizzleClient = PostgresJsDatabase<typeof schema>;

@Global()
@Module({
  providers: [
    {
      provide: DRIZZLE_CLIENT,
      inject: [ConfigService],
      useFactory: (config: ConfigService) => {
        const connectionString = config.getOrThrow<string>("DATABASE_URL");
        const client = postgres(connectionString, {
          max: 10,
          idle_timeout: 30,
          connect_timeout: 10,
          prepare: false, // Supabase pgbouncer transaction mode requires prepare: false
        });
        return drizzle(client, {
          schema,
          logger: process.env.NODE_ENV !== "production",
        });
      },
    },
  ],
  exports: [DRIZZLE_CLIENT],
})
export class DbModule {}
```

## Failure Modes & Recovery

| Symptom                                                              | Likely Cause                           | Fix                                                                                                                                   |
| -------------------------------------------------------------------- | -------------------------------------- | ------------------------------------------------------------------------------------------------------------------------------------- |
| `drizzle-kit generate` errors with "DATABASE_URL not set"            | dotenv not loading                     | Verify `dotenv` installed; `apps/api/.env` exists with DATABASE_URL line                                                              |
| `apply_migration` errors "function gen_random_uuid() does not exist" | pgcrypto extension not enabled         | Apply preliminary migration: `CREATE EXTENSION IF NOT EXISTS pgcrypto;`                                                               |
| `apply_migration` errors "relation already exists"                   | tables already exist (re-run scenario) | Either skip Phase D or drop public schema and retry                                                                                   |
| RLS policy errors "syntax error near 'auth'"                         | Supabase auth schema not present       | Should not happen — Supabase auto-installs auth schema. Check `SELECT * FROM information_schema.schemata WHERE schema_name = 'auth';` |
| Backend type-check errors on Drizzle import                          | tsconfig path alias missing            | Verify `apps/api/tsconfig.json` has `@aurahire/db` path alias                                                                         |
| Backend logs "FATAL: prepared statement already exists"              | postgres client pool config wrong      | `prepare: false` in `db.module.ts` (Supabase pgbouncer transaction mode)                                                              |
| `list_tables` returns empty after migration                          | wrong project_id                       | Verify `fzjvalmouygmmnrgpgtg` is correct via `list_projects`                                                                          |
