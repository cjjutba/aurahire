**Goal:** Ship the **persistent navigation chrome** for every authenticated portal (candidate, recruiter, admin) plus the marketing-page nav/footer foundation, plus the JWT-injection wrapper that lets orval-generated TanStack Query hooks auto-attach the Supabase access token to every backend call. **Tech Stack:** shadcn/ui (`dropdown-menu`, `avatar`, `sheet` for mobile drawer, plus what's already installed), Lucide React icons, Next.js App Router with route groups, Supabase SSR session helpers from Slice 1.4.

**Step 1:** Read current fetcher to see its exact shape.

**Step 2:** Update file. The pattern (preserving the existing `(url, RequestInit)` signature):

```ts
/**
 * Custom fetcher for orval-generated API client.
 * Module-level access token is injected as Authorization: Bearer <jwt>.
 * apps/web's <AuthTokenProvider> calls setAccessToken() on session changes.
 */

let accessToken: string | null = null;

/**
 * Set or clear the access token used for all subsequent API calls.
 * Call with `null` on sign-out.
 */
export function setAccessToken(token: string | null): void {
  accessToken = token;
}

export function getAccessToken(): string | null {
  return accessToken;
}

const getApiUrl = (): string => {
  if (typeof window !== "undefined") {
    const overridden = (window as unknown as { __AURAHIRE_API_URL__?: string })
      .__AURAHIRE_API_URL__;
    if (overridden) return overridden;
  }
  return process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:3333";
};

/**
 * Custom fetch wrapper used as orval's `mutator`.
 * Signature: (url, init) → Promise<T>, matching orval's emitted call style.
 */
export const fetcher = async <T = unknown>(
  url: string,
  init: RequestInit = {},
): Promise<T> => {
  const apiUrl = getApiUrl();
  const fullUrl = url.startsWith("http") ? url : `${apiUrl}${url.startsWith("/") ? url : `/${url}`}`;

  const headers = new Headers(init.headers);
  if (!headers.has("Content-Type") && init.body && typeof init.body === "string") {
    headers.set("Content-Type", "application/json");
  }
  if (accessToken && !headers.has("Authorization")) {
    headers.set("Authorization", `Bearer ${accessToken}`);
  }

  const res = await fetch(fullUrl, { ...init, headers, credentials: "include" });

  if (!res.ok) {
    const errBody = await res.json().catch(() => ({}));
    const err = new Error(`API ${res.status}`) as Error & {
      response: Response;
      body: unknown;
      status: number;
    };
    err.response = res;
    err.body = errBody;
    err.status = res.status;
    throw err;
  }

  // Handle 204 No Content
  if (res.status === 204) return undefined as T;

  return (await res.json()) as T;
};

export type Fetcher = typeof fetcher;
```

### Task 2: Create `apps/web/components/providers/auth-token-provider.tsx`

**Files:**
- Create: `apps/web/components/providers/auth-token-provider.tsx`

**Step 1:** Write file:

```tsx
"use client";

import { useEffect, useMemo, useRef } from "react";
import { setAccessToken } from "@aurahire/shared";
import { createSupabaseBrowserClient } from "@/lib/auth/client";

/**
 * Reads the current Supabase session, sets the module-level access token in the
 * shared API client, and listens for auth state changes (sign-in, sign-out, refresh)
 * to keep the token synchronized.
 *
 * Wraps every page so any orval-generated TanStack Query hook fires with the
 * correct Authorization header automatically.
 */
export function AuthTokenProvider({ children }: { children: React.ReactNode }) {
  const supabase = useMemo(() => createSupabaseBrowserClient(), []);
  const initialized = useRef(false);

  useEffect(() => {
    if (initialized.current) return;
    initialized.current = true;

    // Hydrate from current session (if any)
    void supabase.auth.getSession().then(({ data: { session } }) => {
      setAccessToken(session?.access_token ?? null);
    });

    // Subscribe to auth state changes
    const {
      data: { subscription },
    } = supabase.auth.onAuthStateChange((_event, session) => {
      setAccessToken(session?.access_token ?? null);
    });

    return () => {
      subscription.unsubscribe();
    };
  }, [supabase]);

  return <>{children}</>;
}
```

```tsx
"use client";

import Link from "next/link";
import { usePathname } from "next/navigation";
import { Fragment } from "react";

const SEGMENT_LABELS: Record<string, string> = {
  candidate: "Candidate",
  recruiter: "Recruiter",
  admin: "Admin",
  jobs: "Jobs",
  applications: "Applications",
  interviews: "Interviews",
  profile: "Profile",
  resume: "Resume",
  settings: "Settings",
  candidates: "Candidates",
  shortlist: "Shortlist",
  offers: "Offers",
  analytics: "Analytics",
  users: "Users",
  audit: "Audit",
  "ai-config": "AI Config",
  "bias-monitor": "Bias Monitor",
  new: "New",
  edit: "Edit",
};

export function Breadcrumb() {
  const pathname = usePathname();
  const segments = pathname.split("/").filter(Boolean);

  if (segments.length === 0) return null;

  const crumbs = segments.map((segment, idx) => {
    const href = "/" + segments.slice(0, idx + 1).join("/");
    const label =
      SEGMENT_LABELS[segment] ??
      (segment.length > 20 ? `${segment.slice(0, 8)}...` : segment);
    return { href, label, isLast: idx === segments.length - 1 };
  });

  return (
    <nav aria-label="Breadcrumb" className="text-sm">
      <ol className="flex items-center gap-2">
        {crumbs.map((c, i) => (
          <Fragment key={c.href}>
            {i > 0 && <span className="text-[var(--color-muted-soft)]">/</span>}
            {c.isLast ? (
              <span className="text-[var(--color-ink)]">{c.label}</span>
            ) : (
              <Link
                href={c.href}
                className="text-[var(--color-muted)] transition hover:text-[var(--color-ink)]"
              >
                {c.label}
              </Link>
            )}
          </Fragment>
        ))}
      </ol>
    </nav>
  );
}
```
